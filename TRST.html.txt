<script>
    let currentOption = 1;
    // 方案一 (6.5%) 數據
    const data1 = {
        6: 68880, 8: 85771, 10: 90353, 15: 99323, 20: 112121, 25: 134781, 30: 158491, 35: 184515, 39: 212274
    };
    // 方案二 (10%) 數據
    const data2 = {
        10: 117839, 15: 120164, 20: 122735, 25: 130293, 30: 131693, 35: 126892, 39: 121464
    };

    function switchOption(opt) {
        currentOption = opt;
        document.getElementById('tab1').className = opt === 1 ? 'tab active' : 'tab';
        document.getElementById('tab2').className = opt === 2 ? 'tab active' : 'tab';
        const slider = document.getElementById('year-slider');
        if(opt === 1) {
            slider.min = 6; slider.value = 6;
            document.getElementById('option-desc').innerText = "策略：從第 6 年起，每年提取 HKD 50,700 (佔總保費之 6.5%)";
        } else {
            slider.min = 10; slider.value = 10;
            document.getElementById('option-desc').innerText = "策略：從第 10 年起，每年提取 HKD 78,000 (佔總保費之 10%)";
        }
        updateData();
    }

    function updateData() {
        const year = parseInt(document.getElementById('year-slider').value);
        const dataSet = currentOption === 1 ? data1 : data2;
        const annUSD = currentOption === 1 ? 6500 : 10000;
        
        let keys = Object.keys(dataSet).map(Number);
        let closest = keys.reduce((p, c) => Math.abs(c - year) < Math.abs(p - year) ? c : p);
        let sv = dataSet[closest];

        document.getElementById('year-val').innerText = year;
        document.getElementById('age-val').innerText = 62 + year;
        
        // 該年提取
        document.getElementById('ann-hkd').innerText = `HKD ${(annUSD*7.8).toLocaleString()}`;
        document.getElementById('ann-usd').innerText = `(USD ${annUSD.toLocaleString()})`;
        
        // 累積提取計算
        const cumUSD = currentOption === 1 ? (year - 5) * 6500 : (year - 9) * 10000;
        document.getElementById('cum-hkd').innerText = `HKD ${(cumUSD*7.8).toLocaleString()}`;
        document.getElementById('cum-usd').innerText = `(USD ${cumUSD.toLocaleString()})`;
        
        // 剩餘價值
        document.getElementById('sv-hkd').innerText = `HKD ${(sv*7.8).toLocaleString(undefined, {maximumFractionDigits:0})}`;
        document.getElementById('sv-usd').innerText = `(USD ${sv.toLocaleString()})`;
    }
    updateData();
</script>
